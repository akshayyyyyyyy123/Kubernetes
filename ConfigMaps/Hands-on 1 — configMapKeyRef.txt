STEP 01: Create the ConfigMap

configMap.yaml

apiVersion: v1
kind: ConfigMap
metadata:
  name: app-config
data:
  APP_ENV: "production"
  APP_PORT: "8080"
  LOG_LEVEL: "info"
  APP_NAME: "backend-service"

STEP 02: Create a Pod
configmap-keyref.yaml

apiVersion: v1
kind: Pod
metadata:
  name: configmap-keyref-pod

spec:
  containers:
    - name: app
      image: busybox:1.36

      command:
        - sh
        - -c
        - |
          echo "APP_ENV=$APP_ENV"
          sleep 3600

      env:
        - name: APP_ENV
          valueFrom:
            configMapKeyRef:
              name: app-config
              key: APP_ENV

# kubectl apply -f configmap-keyref.yaml

# kubectl get pods                                                                                                                                    
NAME                           READY   STATUS      RESTARTS      AGE
configmap-keyref-pod           1/1     Running     0             16m

# kubectl exec -it configmap-keyref-pod -- sh
/ # echo $APP_ENV
production
/ # echo $APP_PORT

/ # exit


# kubectl logs configmap-keyref-pod
